🎭 User Types

User

It represents a single iFunny user. Some of the auxiliary types are broken out below.

type User = {
    id: string,
    nick: string,
    original_nick: string,
    about: string,
    
    nick_color: string,
    cover_bg_color: string,
    cover_url: string,
    web_url: string,

    bans: Ban[],
    are_you_blocked: boolean,
    is_available_for_chat: boolean,
    is_banned: boolean,
    is_blocked: boolean,
    is_deleted: boolean,
    is_in_subscribers: boolean,
    is_in_subscriptions: boolean,
    is_private: boolean,
    is_subscribed_to_updates: boolean,
    is_verified: boolean,

    meme_experience: MemeExperience,
    messaging_privacy_status: "closed" | "public" | "subscribers",
    messenger_active: boolean,
    messenger_token: "1010101010101010101010101010101010101010", // deprecated
    num: Numbers,
    photo: ProfilePhoto,
    rating: Rating
}

type Ban = {
    date_until: number, // UNIX timestamp
    id:         string, 
    type:       string
}

type ProfilePhoto = {
	bg_color: string, // hex color code, ie "597aa7"
	url: string,
	thumb: {
		large_url: string,
		medium_url: string,
		small_url: string
	}
}

type Numbers = {
    achievements:  number,
    created:       number,
    featured:      number,
    subscribers:   number,
    subscriptions: number,
    total_posts:   number,
    total_smiles:  number
}

type Numbers = {
    achievements:  number,
    created:       number,
    featured:      number,
    subscribers:   number,
    subscriptions: number,
    total_posts:   number,
    total_smiles:  number
}

type Level = {
	id: string,
	value: number,
	points: number
}

type Rating = {
    points:        number,
    is_show_level: boolean,
    current_level: Level,
    next_level:    Level,
    max_level:     Level
}

Self User

The User type that includes information about the auth'd caller

type SelfUser = User & {
	email:                     string,
	need_account_setup:        boolean,
	safe_mode:                 boolean
    
    is_blocked_in_messenger:      boolean,
	is_ifunny_team_member:        boolean,
	is_moderator:                 boolean,
	have_unnotified_achievements: boolean,
	have_unnotified_bans:         boolean,
	have_unnotified_levels:       boolean,
	have_unnotified_strikes:      boolean,

    
    phone?: string,
    phone_data?: {
        code: string,
        number: string,
    },
    unconfirmed_phone?: string,
    unconfirmed_phone_data?: {
        code: string,
        number: string,
    }
};

Meme Experience

The number of days you've been "active" on iFunny. The exact activity criteria isn't known, but it seems like unused accounts stop accruing rank. The rank determines the badge info.

type MemeExperience = {
  badge_size: { h: number,    w: number },
  badge_url: string,
  days: number,
  next_milestone: number,
  rank: "Meme Explorer"             // 1-4 days
      | "Meme Bro"                  // 5-24 days
      | "Meme Daddy"                // 25-49 days
      | "Dank Memer"                // 50-99 days
      | "Meme Master Baker"         // 100-199 days
      | "Deep Fried Memer"          // 200-299 days
      | "Saucy Memer"               // 300-499 days
      | "Original Meme Gangster"    // 500-665 days
      | "Meme Demon"                // 666-910 days
      | "Steal Beams of Memes"      // 911-999 days
      | "Meme Dealer"               // 1000-1499 days
      | "iFunny Veteran"            // 1500-1999 days
      | "Chef's Meme Agent"         // 2000+ days
